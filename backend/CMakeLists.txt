cmake_minimum_required(VERSION 3.24)
project(AuraStreamBackend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Drogon CONFIG REQUIRED)
find_package(unofficial-sodium CONFIG REQUIRED)
file(GLOB_RECURSE APP_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_executable(aurastream ${APP_SOURCES})
target_include_directories(aurastream PRIVATE include)
target_link_libraries(aurastream PRIVATE Drogon::Drogon unofficial-sodium::sodium)
if(MSVC)
  target_compile_definitions(aurastream PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(aurastream PRIVATE /W4 /utf-8)
endif()
add_executable(queue_tests tests/queue_tests.cpp)
target_include_directories(queue_tests PRIVATE include)
add_test(NAME queue_tests COMMAND queue_tests)
